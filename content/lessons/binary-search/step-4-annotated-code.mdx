<AnnotatedCode
  language="python"
  title="Iterative Binary Search"
  code={`def binary_search(values, target):
    left = 0
    right = len(values) - 1

    while left <= right:
        middle = left + (right - left) // 2

        if values[middle] == target:
            return middle
        elif values[middle] < target:
            left = middle + 1
        else:
            right = middle - 1

    return -1`}
