<AnnotatedCode
  language="python"
  title="Maximum Sum of k Consecutive Values"
  code={`def max_window_sum(values, k):
    if k <= 0 or k > len(values):
        return None

    window_sum = 0
    for index in range(k):
        window_sum += values[index]

    best_sum = window_sum
    for right in range(k, len(values)):
        window_sum += values[right]
        window_sum -= values[right - k]
        best_sum = max(best_sum, window_sum)

    return best_sum`}
